## 一、背景与目标

### 1.1 现状

`createMockFactory` 的 `generateValue` 对 `string` 字段永远生成 `"mock-{field}"`，无语义策略。Agent 需手动填充 `scenarios.default` 中的业务数据。

### 1.2 目标

1. **faker.js 作为可选数据源**：`peerDependency` + 显式 `_faker` 注入（Oracle：删除自动加载，同步 API 无法 `await import()`）
2. **两档策略**：`realistic`（真实感，需 faker）/ `minimal`（最小化，现有逻辑）（Oracle：删除 `random`——无分派分支）
3. **字段名→策略自动映射**：内置三层匹配表 + `CUSTOM_STRATEGIES` 合并为单一分派表（Oracle 🔴2）
4. **可插拔数据源**：`_faker` 参数注入，`_generators` 自定义生成器
5. **codegen 预填充**：`scenariosSkeleton` 调用 `makeN()` 自动填充，`factorySkeleton` 提前声明（Oracle 🔴4 TDZ）
6. **枚举轮转**：`cycleNext()` 每工厂闭包内持有（Oracle 🔴4 全局→per-factory）

### 1.3 不做什么

- ❌ 不内置 faker.js（peerDependency，消费端决定是否安装）
- ❌ 不做自动加载（Oracle：同步 API 不支持 `await import()`，仅支持显式 `_faker` 注入）
- ❌ 不改现有 `minimal` 策略行为（向后兼容）
- ❌ 不做 `_relations` 关联关系生成（拆至 v2.0.1）

### 3.1 MockFactoryOptions 扩展

```typescript
export type GenerationStrategy = "minimal" | "realistic";

export interface GeneratorConfig {
  /** 策略名（如 "person.fullName" / "company.name" / "phone.number"） */
  strategy?: string;
  /** 数值范围（仅对 number 类字段有效） */
  min?: number;
  max?: number;
  /** 自定义生成函数 */
  generator?: (ctx: GeneratorContext) => unknown;
}

export interface GeneratorContext {
  fieldName: string;
  strategy: GenerationStrategy;
  /** 可变状态引用（Oracle 🔴3：确保跨 make 持久） */
  state: MutableFactoryState;
  depth: number;
  maxDepth: number;
  dateBase: Date;
  enumValues?: readonly (string | number)[];
  faker?: Faker;  // 外部注入，不含自动加载
  /** 字段级生成器覆盖 */
  generators?: Record<string, GeneratorConfig>;
  /** 枚举值映射 */
  enums?: Record<string, readonly (string | number)[]>;
}

export interface MutableFactoryState {
  seed: number;       // LCG 种子（跨 make 持久）
  counter: number;    // make 调用计数（每 make 递增）
  idCounter: number;  // isId 自增
  cycleState: Map<string, number>;  // 枚举轮转状态（per-factory，Oracle 🔴4）
}

export interface MockFactoryOptions<T> {
  _types?: Record<keyof T, MockFieldSchema> | MockFieldSchema;
  _enums?: Record<string, readonly (string | number)[]>;
  _seed?: number;
  _maxDepth?: number;
  _dateBase?: Date;
  /** 生成策略（默认 "minimal"） */
  _strategy?: GenerationStrategy;
  /** 字段级生成策略覆盖 */
  _generators?: Record<string, GeneratorConfig>;
  /** 外部 faker 实例（仅显式注入，Oracle 🔴6：删除自动加载） */
  _faker?: Faker;
  // 注意：_relations 已拆至 v2.0.1
}
```

### 3.2 统一字段名→策略分派表（Oracle 🔴2 修复）

```typescript
// 统一解析入口：先查 CUSTOM → 精确 → 后缀 → 前缀
function resolveFieldStrategy(field: string): string | null {
  // 0. CUSTOM_STRATEGIES 键名（精确字段名）
  if (CUSTOM_STRATEGIES[field]) return field;
  // 1. 精确匹配
  if (EXACT_STRATEGIES[field]) return EXACT_STRATEGIES[field];
  // 2. 后缀匹配
  for (const [suffix, strategy] of Object.entries(SUFFIX_STRATEGIES)) {
    if (field.endsWith(suffix)) return strategy;
  }
  // 3. 前缀匹配
  for (const [prefix, strategy] of Object.entries(PREFIX_STRATEGIES)) {
    if (field.startsWith(prefix)) return strategy;
  }
  // 4. 字段名模式：*Time / *Id
  if (/Time$|At$/.test(field)) return "__date__";
  if (/Id$/.test(field)) return "__id__";
  return null;
}
```

**CUSTOM_STRATEGIES**（faker 不擅长的字段，合并到同一分派表使可路由）：

```typescript
const CUSTOM_STRATEGIES: Record<string, (ctx: GeneratorContext) => unknown> = {
  // 金额
  amount: (ctx) => numberInRange(ctx, 1, 99999),
  price: (ctx) => numberInRange(ctx, 10, 9999),
  total: (ctx) => numberInRange(ctx, 100, 99999),
  quantity: (ctx) => numberInRange(ctx, 1, 999),
  discount: (ctx) => numberInRange(ctx, 0, 100) / 100,
  rate: (ctx) => numberInRange(ctx, 0, 100) / 100,
  score: (ctx) => numberInRange(ctx, 1, 5),

  // 状态枚举轮转
  status: (ctx) => cycleNext(ctx.state, ctx.fieldName, ctx.enums?.[ctx.fieldName] ?? ctx.enumValues ?? []),
  state: (ctx) => cycleNext(ctx.state, ctx.fieldName, ctx.enums?.[ctx.fieldName] ?? ctx.enumValues ?? []),
  level: (ctx) => cycleNext(ctx.state, ctx.fieldName, ctx.enums?.[ctx.fieldName] ?? ctx.enumValues ?? []),

  // 时间轴递增
  createdAt: (ctx) => advanceDate(ctx),
  updatedAt: (ctx) => advanceDate(ctx),
  deletedAt: null,  // null 表示不生成（可选字段）

  // ID 自增（按 schema.kind 决定 string 或 number）
  __id__: (ctx) => ctx.enumValues === undefined ? ++ctx.state.idCounter : `mock-${++ctx.state.idCounter}`,
  __date__: (ctx) => advanceDate(ctx),
};
```

### 3.3 枚举轮转器（per-factory，Oracle 🔴4）

```typescript
function cycleNext(
  cycleState: Map<string, number>,
  field: string,
  values: readonly (string | number)[],
): string | number {
  if (values.length === 0) return "ACTIVE";
  const state = cycleState.get(field) ?? 0;
  const value = values[state % values.length];
  cycleState.set(field, state + 1);
  return value;
}
```

### 3.4 策略分派逻辑（Oracle 🔴2/🔴3 修复）

```typescript
function generateValue(
  field: string, schema: MockFieldSchema, depth: number, ctx: GeneratorContext,
): unknown {
  if (depth > ctx.maxDepth) return undefined;  // Oracle 🔴3：补 maxDepth

  // 1. _generators 覆盖（优先级最高，不限策略）
  const genConfig = ctx.generators?.[field];
  if (genConfig?.generator) return genConfig.generator(ctx);
  if (genConfig?.strategy) return applyFakerMethod(ctx, genConfig.strategy);

  // 2. realistic 策略：字段名映射
  if (ctx.strategy === "realistic" && ctx.faker) {
    const strategy = resolveFieldStrategy(field);
    if (strategy === "__id__") {
      // ID 自增：按 schema.kind 决定 string/number（Oracle 🟡4）
      const counter = ++ctx.state.idCounter;
      return schema.kind === "string" ? `mock-${counter}` : counter;
    }
    if (strategy === "__date__") return advanceDate(ctx);
    if (strategy && CUSTOM_STRATEGIES[strategy]) return CUSTOM_STRATEGIES[strategy](ctx);
    if (strategy) return applyFakerMethod(ctx, strategy);
    return generateByKind(schema, ctx);  // 按 kind 回退
  }

  // 3. minimal：现有逻辑
  return generateMinimal(field, schema, ctx);
}
```
